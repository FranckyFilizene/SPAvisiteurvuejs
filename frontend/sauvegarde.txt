<template>
  <div class="w-full flex items-center justify-center min-h-screen p-6">
    <!-- Section Illustration -->
    <div class="flex-1 hidden md:flex justify-center items-center">
      <img src="/Secure login-pana.svg" class="w-full max-w-lg" alt="Illustration de connexion" />
    </div>

    <!-- Section Formulaire -->
    <div class="flex-1 flex flex-col justify-center items-center md:items-start">
      <div class="mb-6 w-full max-w-md">
        <div class="badge badge-primary mb-2">visiteur.com</div>
        <h1 class="text-5xl md:text-6xl font-bold">Connexion</h1>
      </div>

      <form @submit.prevent="handleSubmit" class="w-full max-w-md bg-base-100 p-6 rounded-box shadow-lg border border-base-200">
        
        <!-- Alerte d'erreur -->
        <div v-if="errorMessage" class="alert alert-error mb-4 py-2 text-sm text-white">
          <span>{{ errorMessage }}</span>
        </div>

        <!-- Champ Email / Téléphone -->
        <div class="form-control w-full mb-4">
          <label class="label">
            <span class="label-text font-semibold">Email ou téléphone</span>
          </label>
          <input 
            v-model="form.identifier"
            type="text" 
            placeholder="Ex: admin@gmail.com" 
            class="input input-bordered w-full"
            :disabled="isLoading"
            required
          />
        </div>

        <!-- Champ Mot de passe -->
        <div class="form-control w-full mb-4">
          <label class="label">
            <span class="label-text font-semibold">Mot de passe</span>
          </label>
          <div class="relative flex items-center">
            <input 
              v-model="form.password"
              :type="showPassword ? 'text' : 'password'" 
              placeholder="••••••••"
              class="input input-bordered w-full pr-10"
              :disabled="isLoading"
              required
            />
            <button 
              type="button" 
              @click="togglePasswordVisibility" 
              class="absolute right-3 text-base-content/60 hover:text-base-content transition-colors"
            >
              <v-icon :name="showPassword ? 'fa-eye-slash' : 'fa-eye'" />
            </button>
          </div>
        </div>

        <!-- Options supplémentaires -->
        <div class="flex w-full justify-between items-center my-4 text-sm">
          <label class="label cursor-pointer gap-2 p-0">
            <input 
              v-model="form.rememberMe"
              type="checkbox" 
              class="checkbox checkbox-sm checkbox-primary" 
              :disabled="isLoading"
            />
            <span class="label-text">Se souvenir de moi</span>
          </label>
          <router-link to="/mot-de-passe-oublie" class="link link-hover link-primary">
            Mot de passe oublié ?
          </router-link>
        </div>

        <!-- Boutons d'action -->
        <div class="mt-6 flex flex-col sm:flex-row gap-3">
          <button type="submit" class="btn btn-primary flex-1" :disabled="isLoading">
            <span v-if="isLoading" class="loading loading-spinner loading-sm"></span>
            <span v-else>Se connecter</span>
          </button>
          <router-link to="/inscription" class="btn btn-outline flex-1">
            S'inscrire
          </router-link>
        </div>

        <div class="divider my-6">OU</div>
      </form>
    </div>
  </div>
</template>

<script>
// 1. Import d'Axios
import axios from 'axios';

export default {
  name: 'LoginView',
  data() {
    return {
      showPassword: false,
      isLoading: false,       // Pour afficher l'animation de chargement sur le bouton
      errorMessage: null,     // Pour stocker et afficher les erreurs API
      form: {
        identifier: '',
        password: '',
        rememberMe: false
      }
    }
  },

  methods: {
    togglePasswordVisibility() {
      this.showPassword = !this.showPassword;
    },

    // 2. Méthode asynchrone pour l'appel Axios
    async handleSubmit() {
      this.isLoading = true;
      this.errorMessage = null;

      try {
        // Envoi des données vers l'API backend
        const response = await axios.post('https://api.votre-site.com/login', {
          identifier: this.form.identifier,
          password: this.form.password,
          remember_me: this.form.rememberMe
        });

        // Succès : Récupération du Token d'authentification
        const token = response.data.token;
        
        // Stockage du token (localStorage ou sessionStorage selon 'Se souvenir de moi')
        if (this.form.rememberMe) {
          localStorage.setItem('user_token', token);
        } else {
          sessionStorage.setItem('user_token', token);
        }

        // Redirection vers le tableau de bord
        this.$router.push('/dashboard');

      } catch (error) {
        // Gestion des erreurs HTTP (Ex: 401 Identifiants incorrects, 500 Erreur serveur)
        if (error.response) {
          this.errorMessage = error.response.data.message || 'Identifiants invalides.';
        } else if (error.request) {
          this.errorMessage = 'Impossible de contacter le serveur. Vérifiez votre connexion.';
        } else {
          this.errorMessage = 'Une erreur inattendue est survenue.';
        }
      } finally {
        // Désactivation de l'état de chargement dans tous les cas
        this.isLoading = false;
      }
    }
  }
}
</script>
